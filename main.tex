\begin{exercise}[ID=caida, subtitle={Caída libre}, tags={examen}, points=50]
Una piedra se suelta desde el reposo a una altura $h=\qty{20}{\m}$. Despreciando la fricción con el aire:
\begin{enumerate}
    \item Calcule la rapidez con que llega al suelo. Tome $g=\qty{9.8}{\m/\s^2}$.
    \item ¿Depende el resultado de la masa de la piedra? Explique.
\end{enumerate}

\vspace{5cm}
\end{exercise}
\begin{solution}
\begin{enumerate}
    \item Por conservación de la energía mecánica,
    \begin{align}
        \cancel{m}gh = \frac{1}{2}\cancel{m}v^2
        \;\Longrightarrow\;
        v = \sqrt{2gh} = \sqrt{2\times 9.8\times 20}\,\unit{\m/\s} \approx \qty{19.8}{\m/\s}\,.
    \end{align}
    \item No: la masa se cancela en los dos lados de la ecuación.
\end{enumerate}
\end{solution}

\begin{exercise}[ID=choque, subtitle={Acción y reacción}, tags={examen}, points=50]
Un carro pequeño choca de frente contra un camión pesado. Durante el choque, ¿cuál de los siguientes enunciados es correcto?
\begin{enumerate}
    \item La fuerza del camión sobre el carro es mayor que la del carro sobre el camión.
    \item La fuerza del carro sobre el camión es mayor que la del camión sobre el carro.
    \item Las dos fuerzas tienen la misma magnitud y direcciones opuestas.
    \item Depende de cuál de los dos vehículos iba más rápido.
\end{enumerate}
\begin{comment}
% Parte que no se pregunta en este examen: no se imprime, ni aun con \solucionestrue.
(II) ¿Cuál de los dos vehículos sufre el mayor cambio de velocidad?
\end{comment}
\end{exercise}
\begin{solution}
c). Por acción y reacción, las fuerzas que se hacen dos cuerpos son iguales en magnitud y opuestas, sin importar sus masas ni sus velocidades.
\end{solution}

\begin{exercise}[ID=impulso, subtitle={Fuerza media del bate}, tags={banco}, points=25]
Una pelota de béisbol de $\qty{0.15}{\kilo\g}$ llega al bate con una rapidez de $\qty{40}{\m/\s}$ y sale en la dirección contraria con la misma rapidez. El contacto dura $\Delta t=\qty{2}{\milli\s}$. Calcule la fuerza media que hace el bate usando la ley de la fuerza, $F=\Delta p/\Delta t$.
\end{exercise}
\begin{solution}
Tomando como positivo el sentido de salida,
\begin{align}
    F = \frac{\Delta p}{\Delta t} = \frac{(0.15)(40)-(0.15)(-40)}{\num{2e-3}}\,\unit{\N} = \qty{6000}{\N}\,.
\end{align}
\end{solution}
